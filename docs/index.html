Usage

Basic Usage Example

use GhostAuth\GhostAuthManager;
use GhostAuth\Providers\EmailPasswordProvider;

$authManager = new GhostAuthManager();

$result = $authManager->authenticate(
    new EmailPasswordProvider(),
    [
        'email' => 'user@example.com',
        'password' => 'password123',
    ]
);

if ($result->isValid()) {
    // Authentication successful
    $user = $result->getUser();
    // Set session, etc.
} else {
    // Handle errors
    $errors = $result->getErrors();
}

Basic Configuration

// Example config
$config = [
    'cookie' => [
        'name' => 'ghostauth_session',
        'path' => '/',
        'domain' => null,
        'secure' => true,
        'httponly' => true,
        'samesite' => 'Strict',
    ],
    'jwt' => [

        'algorithm' => 'RS256',
        'ttl' => 3600,
    ],
    'providers' => [
        'email' => [
            'model' => 'App\Models\User',
            'table' => 'users',
        ],
        'google' => [
            'client_id' => 'your-google-client-id',
            'client_secret' => 'your-google-client-secret',
            'redirect_uri' => '/auth/google/callback',
        ],
    ],
];
